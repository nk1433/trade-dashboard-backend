//@version=6
indicator("Stockbee signals", overlay=true)

// ==========================================
// 1. INPUTS & SETTINGS
// ==========================================
grp_ma = "Moving Average Settings"
shortLen = input.int(7, "Short MA (avgc7)", group=grp_ma)
longLen  = input.int(65, "Long MA (avgc65)", group=grp_ma)

grp_ui = "Signal & Visual Settings"
threshold    = input.float(4.0, "Percentage Threshold", step=0.1, group=grp_ui)
tightRange   = input.float(0.5, "Tight Bar Range (+/-%)", step=0.1, group=grp_ui)

// --- Expanded Panel Controls ---
showPanelUp   = input.bool(false, "Show Panel: Up 4% Labels", group=grp_ui)
showPanelDown = input.bool(false, "Show Panel: Down 4% Labels", group=grp_ui)
showPanelFade = input.bool(true, "Show Panel: Fadeout Labels", group=grp_ui) // Controls historical fadeout markers

grp_study = "Latest Bar Study"
showLiveCS = input.bool(true, "Show Live Closing Strength %", group=grp_study)

// ==========================================
// 2. CORE CALCULATIONS
// ==========================================
pctChange = (close - close[1]) / close[1] * 100
barRange  = high - low
closingStrength = barRange != 0 ? (close - low) / barRange * 100 : na

shortMA = ta.sma(close, shortLen)
longMA  = ta.sma(close, longLen)
volCondition = ta.lowest(volume[1], 3) >= 100000

// ==========================================
// 3. SIGNAL LOGIC
// ==========================================

// --- SLTB / Trend Intensity ---
ti_maCondition  = shortMA / longMA >= 1.05
ti_bullishBar   = close > open and close > close[1]
ti_acceleration = (close / close[1] > close[1] / close[2]) and (close[1] / close[2] < 1.02)
sltb_signal     = volCondition and (close >= 25) and ti_maCondition and ti_bullishBar and ti_acceleration

// --- Dollar BO ($50+ move) ---
dollarBO_signal = (close - open >= 50) and (volume > 100000)

// --- Up >= 4% ---
threeUpInARow = close[1] > close[2] and close[2] > close[3] and close[3] > close[4]
upBar = pctChange >= threshold and not threeUpInARow

// --- Down <= -4% ---
downBar = pctChange <= -threshold

// --- Fadeout Criteria ---
// Volume < Prev Volume AND Range < Prev Range
isFadeoutRaw = (volume < volume[1]) and (barRange < barRange[1])
bullish_combined = sltb_signal or upBar or dollarBO_signal

// --- UPDATED ANTS LOGIC (MDT & DTS Variations Included) ---
// 1. Original TI: avgc7/avgc65 > 1.05
ti_orig = (shortMA / longMA >= 1.05)
// 2. MDT: c/avgc126 > 1.19
ti_mdt  = (close / ta.sma(close, 126) > 1.19)
// 3. DTS: c/minl252 >= 1.8
ti_dts  = (close / ta.lowest(low, 252) >= 1.8)

// Matches ANY of the 3 variations + Volume + Tight Range
showTight = (ti_orig or ti_mdt or ti_dts) and volCondition and (math.abs(pctChange) <= tightRange)

// ==========================================
// 4. LATEST BAR STUDY (Live CS Only)
// ==========================================
var label liveLabel = na
if showLiveCS and barstate.islast
    label.delete(liveLabel)
    lColor = closingStrength >= 70 ? color.green : color.red
    // Display simple CS% label on the current bar
    liveLabel := label.new(bar_index, high, 
         text=str.tostring(math.round(closingStrength, 1)) + "%", 
         color=lColor, 
         textcolor=color.white, 
         style=label.style_label_up, 
         yloc=yloc.belowbar)

// ==========================================
// 5. DYNAMIC SMART ALERT
// ==========================================
if barstate.isconfirmed
    string sigType = ""
    if dollarBO_signal
        sigType := "Dollar BO"
    else if sltb_signal
        sigType := "SLTB"
    else if upBar
        sigType := "Up 4%"
    else if downBar
        sigType := "Down 4%"

    if sigType != ""
        string alertMsg = "Stockbee signals: " + sigType + "! | " + syminfo.ticker + 
             " | Price: " + str.tostring(close) + 
             " | Day Range: " + str.tostring(open) + " to " + str.tostring(close) + 
             " | Volume: " + str.tostring(volume, format.volume)
        
        alert(alertMsg, alert.freq_once_per_bar_close)

// ==========================================
// 6. MANUAL DROPDOWN ALERTS (UI)
// ==========================================
msgTemplate = " | {{ticker}} | Price: {{close}} | Vol: {{volume}} | Time: {{time}}"

alertcondition(sltb_signal,     "SLTB Alert",      "SLTB Alert"      + msgTemplate)
alertcondition(dollarBO_signal, "Dollar BO Alert", "Dollar BO Alert" + msgTemplate)
alertcondition(upBar,           "Up 4% Alert",     "Up 4% Alert"     + msgTemplate)
alertcondition(downBar,         "Down 4% Alert",   "Down 4% Alert"   + msgTemplate)
alertcondition(bullish_combined, "Combined Bullish (SLTB/Up/Dollar)", "Bullish Signal Triggered" + msgTemplate)

// ==========================================
// 7. VISUAL PLOTTING
// ==========================================
// Core Shapes (Always on if condition is met)
plotshape(sltb_signal,     title="SLTB",      style=shape.circle,     location=location.belowbar, color=color.blue,   size=size.tiny)
plotshape(dollarBO_signal, title="Dollar BO", style=shape.diamond,    location=location.belowbar, color=color.purple, size=size.tiny)
plotshape(showTight,       title="Ants",      style=shape.circle,     location=location.belowbar, color=color.orange, size=size.tiny)

// --- CONDITIONAL PANELS (Visuals binded to inputs) ---

// 1. Up 4% Panel (Green Circle + Label)
plotshape(upBar, title="Up 4%", style=shape.circle, location=location.belowbar, color=color.green, size=size.tiny)
if showPanelUp and upBar and not barstate.islast
    label.new(bar_index, low, str.tostring(math.round(closingStrength, 1)) + "%", yloc=yloc.belowbar, style=label.style_label_up, color=color.new(color.green, 0), textcolor=color.white, size=size.small)

// 2. Down 4% Panel (Red Circle + Label)
plotshape(downBar, title="Down 4%", style=shape.circle, location=location.belowbar, color=color.red, size=size.tiny)
if showPanelDown and downBar and not barstate.islast
    label.new(bar_index, high, str.tostring(math.round(closingStrength, 1)) + "%", yloc=yloc.abovebar, style=label.style_label_down, color=color.new(color.red, 0), textcolor=color.white, size=size.small)

// 3. Fadeout Panel (Yellow Triangle)
plotshape(showPanelFade and isFadeoutRaw, title="Fadeout", style=shape.triangleup, location=location.belowbar, color=color.yellow, size=size.tiny)
